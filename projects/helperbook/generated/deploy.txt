// === Dockerfile ===
FROM rust:1.83-slim AS builder
WORKDIR /app
COPY Cargo.toml Cargo.lock ./
RUN mkdir src && echo "fn main() {}" > src/main.rs && cargo build --release && rm -rf src
COPY src/ src/
RUN touch src/main.rs && cargo build --release

FROM debian:bookworm-slim
RUN apt-get update && apt-get install -y ca-certificates && rm -rf /var/lib/apt/lists/*
COPY --from=builder /app/target/release/axis-server /usr/local/bin/axis-server
WORKDIR /app
EXPOSE 8080
CMD ["axis-server"]
// === docker-compose.yml ===
services:
  app:
    build: .
    ports:
      - "8080:8080"
    environment:
      - PORT=8080
      - JWT_SECRET=${JWT_SECRET:-change-me-in-production}
      - RUN_MIGRATIONS=1
      - DATABASE_URL=postgres://axis:axis@postgres:5432/axis
      - SMS_URL=sms_service_url
      - GOOGLE_AUTH_URL=google_auth_service_url
      - APPLE_AUTH_URL=apple_auth_service_url
      - LINKEDIN_AUTH_URL=linkedin_auth_service_url
      - META_AUTH_URL=meta_auth_service_url
      - TOTP_URL=totp_service_url
      - PAYMENT_URL=payment_service_url
      - OPENAI_URL=openai_service_url
      - SEARCH_URL=search_service_url
      - GEO_URL=geo_service_url
      - SEARCH_PROVIDERS_URL=search_service_url
      - AI_MODERATION_URL=openai_service_url
      - CALENDAR_URL=calendar_service_url
      - EMAIL_URL=email_service_url
      - PUSH_URL=push_service_url
      - STORAGE_URL=storage_service_url
    depends_on:
      postgres:
        condition: service_healthy

  postgres:
    image: postgres:16-alpine
    environment:
      - POSTGRES_USER=axis
      - POSTGRES_PASSWORD=axis
      - POSTGRES_DB=axis
    ports:
      - "5432:5432"
    volumes:
      - pgdata:/var/lib/postgresql/data
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U axis"]
      interval: 5s
      timeout: 5s
      retries: 5

volumes:
  pgdata:

  name: axis-app

    matchLabels:
      app: axis-app
  template:
    metadata:
      labels:
        app: axis-app
    spec:
      containers:
        - name: axis-app
          image: axis-server:latest
          ports:

          env:
            - name: PORT

            - name: DATABASE_URL
              valueFrom:
                secretKeyRef:
                  name: axis-secrets
                  key: database-url
          readinessProbe:

              path: /healthz
              port: 8080

            periodSeconds: 10
